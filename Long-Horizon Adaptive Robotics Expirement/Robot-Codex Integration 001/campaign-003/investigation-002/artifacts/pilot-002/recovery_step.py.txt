import json, subprocess, sys, time
from pathlib import Path

EXPECTED=(6,5)

def cli(args):
    r=subprocess.run(['python3','-B','robot.py']+args,capture_output=True,text=True)
    try: obj=json.loads(r.stdout)
    except Exception:
        print(json.dumps({'transport_unknown':True,'returncode':r.returncode,'stdout':r.stdout,'stderr':r.stderr}),flush=True)
        raise SystemExit(4)
    return r.returncode,obj

def terminal(rid, initial):
    obj=initial
    end=time.monotonic()+15
    while obj.get('command',{}).get('phase')!='terminal':
        if time.monotonic()>end:
            print(json.dumps({'unresolved_request':rid,'last_reply':obj}),flush=True)
            raise SystemExit(4)
        time.sleep(0.25)
        code,obj=cli(['query',rid])
        if code or not obj.get('ok'):
            print(json.dumps(obj),flush=True)
            raise SystemExit(code or 4)
    return obj

def brief(s):
    p=s.get('packet',{})
    return {**{k:s.get(k) for k in ('ok','owner','authority_epoch','order_version','observation_sequence','observation_age_s','active_command','robot_action_budget')},'packet':{k:p.get(k) for k in ('capture_sim_time_s','tcp_position_m','gripper','detections','station','joint_velocities')}}

def run_step(rid,args):
    code,before=cli(['status'])
    if code or before.get('owner')!='robot' or (before.get('authority_epoch'),before.get('order_version'))!=EXPECTED or before.get('active_command'):
        print(json.dumps({'precondition_failed':brief(before)}),flush=True)
        raise SystemExit(3)
    code,reply=cli(args+['--request-id',rid])
    Path('notes/'+rid+'-admission.json').write_text(json.dumps(reply))
    if code or not reply.get('ok'):
        print(json.dumps(reply),flush=True)
        raise SystemExit(code or 3)
    reply=terminal(rid,reply)
    print(json.dumps({'request_id':rid,'phase':reply.get('command',{}).get('phase'),'status':reply.get('command',{}).get('status'),'feedback':reply.get('command',{}).get('feedback')}),flush=True)
    if reply.get('command',{}).get('status')!='completed':
        raise SystemExit(3)
    capture_id=rid+'-obs'
    code,obs=cli(['observe','--request-id',capture_id])
    if code or not obs.get('ok'):
        print(json.dumps(obs),flush=True)
        raise SystemExit(code or 3)
    obs=terminal(capture_id,obs)
    code,s=cli(['status'])
    record={'receipt':reply,'capture':obs,'status':s}
    Path('notes/'+rid+'-result.json').write_text(json.dumps(record))
    print(json.dumps({'capture_sequence':obs.get('command',{}).get('observation_sequence'),'status':brief(s)}),flush=True)
    if code or s.get('owner')!='robot' or (s.get('authority_epoch'),s.get('order_version'))!=EXPECTED or s.get('active_command'):
        raise SystemExit(3)
    return s

if __name__=='__main__':
    run_step(sys.argv[1],sys.argv[2:])
